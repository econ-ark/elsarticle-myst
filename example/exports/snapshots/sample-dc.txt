"""Solve ax^2 + bx + c = 0 using the quadratic formula."""
discriminant = b**2 - 4*a*c
if discriminant < 0:
raise ValueError("No real solutions")
x1 = ( -b + np.sqrt(discriminant)) / (2*a)
x2 = ( -b - np.sqrt(discriminant)) / (2*a)
return x1, x2
# Example usage
