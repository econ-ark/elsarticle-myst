%% LaTeX escaping of plain-text frontmatter fields (short_title, credit
%% roles, author notes, affiliation strings) is applied inline via Jinja
%% `replace` chains. The order of replacements matters: backslash and
%% braces are escaped first so later replacements that introduce
%% backslashes do not get re-escaped. Fields that pass through MyST's
%% markdown AST (title, subtitle, abstract, content) are NOT escaped
%% here; the AST already handles LaTeX special characters in those
%% contexts.

[# if options.document_class == 'elsarticle' and options.journal #]
%% Backslash goes to a SENTINEL first and is expanded LAST. Escaping it up
%% front emits \textbackslash{}, whose braces the later brace rules then escape
%% into \textbackslash\{\}, which typesets as "\{}" instead of "\".
\journal{[-options.journal | replace("\\", "@BSL@") | replace("{", "\\{") | replace("}", "\\}") | replace("&", "\\&") | replace("%", "\\%") | replace("$", "\\$") | replace("#", "\\#") | replace("_", "\\_") | replace("~", "\\textasciitilde{}") | replace("^", "\\textasciicircum{}") | replace("@BSL@", "\\textbackslash{}")-]}
[# endif #]

%% \fnmark needs disjoint ranges: [1..N] per-author notes by loop.index, [N+1]
%% the shared equal-contributor note, for N authors.
[# set author_count = doc.authors | length if doc.authors else 0 #]
[# set eq_fnmark = author_count + 1 #]
[# set has_equal = doc.authors and (doc.authors | selectattr("equal_contributor") | list | length > 1) #]

%% namespace shared with affiliations, so bare integers collide: \tnoteref{1}
%% would render the affiliation mark. Prefixes are empty for CAS, which is safe.
[# set tnp = 'tn' if options.document_class == 'elsarticle' else '' #]
[# set corp = 'cor' if options.document_class == 'elsarticle' else '' #]
[# set fnp = 'fn' if options.document_class == 'elsarticle' else '' #]

[# else #]
\shorttitle{[-doc.title | replace("\\", "@BSL@") | replace("{", "\\{") | replace("}", "\\}") | replace("&", "\\&") | replace("%", "\\%") | replace("$", "\\$") | replace("#", "\\#") | replace("_", "\\_") | replace("~", "\\textasciitilde{}") | replace("^", "\\textasciicircum{}") | replace("@BSL@", "\\textbackslash{}")-]}
[# endif #]

\shortauthors{[# for author in doc.authors #][-author.name.split(" ") | last | replace("\\", "@BSL@") | replace("{", "\\{") | replace("}", "\\}") | replace("&", "\\&") | replace("%", "\\%") | replace("$", "\\$") | replace("#", "\\#") | replace("_", "\\_") | replace("~", "\\textasciitilde{}") | replace("^", "\\textasciicircum{}") | replace("@BSL@", "\\textbackslash{}")-][# if not loop.last #], [# endif #][# endfor #]}
[# endif #]

[# if options.document_class == 'elsarticle' #]

[# for author in doc.authors #]
[#- set author_opts = '' -#]
[#- if author.orcid -#][#- set author_opts = 'orcid=' ~ author.orcid -#][#- endif -#]
[#- if author.deceased -#][#- set author_opts = author_opts ~ (',' if author_opts else '') ~ 'deceased' -#][#- endif -#]
[#- if author.facebook -#][#- set author_opts = author_opts ~ (',' if author_opts else '') ~ 'facebook=' ~ author.facebook -#][#- endif -#]
[#- if author.linkedin -#][#- set author_opts = author_opts ~ (',' if author_opts else '') ~ 'linkedin=' ~ author.linkedin -#][#- endif -#]
[#- if author.twitter -#][#- set author_opts = author_opts ~ (',' if author_opts else '') ~ 'twitter=' ~ author.twitter -#][#- endif -#]
%% Two CAS-only shapes here. The trailing [orcid=...] group: elsarticle's \author
%% takes only the affiliation label, so a second group prints as literal text.
%% The marks: elsarticle wants \corref/\fnref INSIDE the name group, not after.
[# if options.document_class == 'elsarticle' #]
\author[[- author.affiliations | join(",", "index") -]]{[-author.name-][# if author.corresponding #]\corref{[-corp-][-loop.index-]}[# endif #][# if author.note #]\fnref{[-fnp-][-loop.index-]}[# endif #][# if author.equal_contributor and has_equal #]\fnref{[-fnp-][-eq_fnmark-]}[# endif #]}
[# else #]
\author[[- author.affiliations | join(",", "index") -]]{[-author.name-]}[# if author_opts #][[- author_opts -]][# endif #]
[# if author.corresponding #]
\cormark[[-loop.index-]]
[# endif #]
[# if author.note #]
\fnmark[[-loop.index-]]
[# endif #]
[# if author.equal_contributor and has_equal #]
\fnmark[[-eq_fnmark-]]
[# endif #]
[# endif #]
[# if author.email #]

[# endif #]

            state={[-affiliation.state | replace("\\", "@BSL@") | replace("{", "\\{") | replace("}", "\\}") | replace("&", "\\&") | replace("%", "\\%") | replace("$", "\\$") | replace("#", "\\#") | replace("_", "\\_") | replace("~", "\\textasciitilde{}") | replace("^", "\\textasciicircum{}") | replace("@BSL@", "\\textbackslash{}")-]}[# endif #][# if affiliation.country #],

%% `keyword' (singular) in elsarticle, `keywords' (plural) in CAS; both take

[# if doc.keywords or parts.jel_codes #]

%% elsarticle's \JEL is an argument-less SWITCH (label, then following text);
