%% LaTeX escaping of plain-text frontmatter fields (short_title, credit
%% roles, author notes, affiliation strings) is applied inline via Jinja
%% `replace` chains. The order of replacements matters: backslash and
%% braces are escaped first so later replacements that introduce
%% backslashes do not get re-escaped. Fields that pass through MyST's
%% markdown AST (title, subtitle, abstract, content) are NOT escaped
%% here; the AST already handles LaTeX special characters in those
%% contexts.

%% --- Footnote-mark index allocation -----------------------------------
%% CAS uses two independent mark spaces (\cormark and \fnmark), so they
%% cannot collide with each other. We allocate disjoint integer ranges
%% inside the \fnmark space:
%%   [1 .. N]                   per-author notes (loop.index)
%%   [N+1]                      shared equal-contributor note
%% where N is the author count. \cormark uses its own [1..]
%% sequence indexed by corresponding-author order.
[# set author_count = doc.authors | length if doc.authors else 0 #]
[# set eq_fnmark = author_count + 1 #]
[# set has_equal = doc.authors and (doc.authors | selectattr("equal_contributor") | list | length > 1) #]

[# if doc.short_title #]
\shorttitle{[-doc.short_title | replace("\\", "\\textbackslash{}") | replace("{", "\\{") | replace("}", "\\}") | replace("&", "\\&") | replace("%", "\\%") | replace("$", "\\$") | replace("#", "\\#") | replace("_", "\\_") | replace("~", "\\textasciitilde{}") | replace("^", "\\textasciicircum{}")-]}
[# else #]
\shorttitle{[-doc.title | replace("\\", "\\textbackslash{}") | replace("{", "\\{") | replace("}", "\\}") | replace("&", "\\&") | replace("%", "\\%") | replace("$", "\\$") | replace("#", "\\#") | replace("_", "\\_") | replace("~", "\\textasciitilde{}") | replace("^", "\\textasciicircum{}")-]}
[# endif #]

%% Extract surname as last word from full name (handles middle initials correctly)
[# if doc.authors #]
\shortauthors{[# for author in doc.authors #][-author.name.split(" ") | last | replace("\\", "\\textbackslash{}") | replace("{", "\\{") | replace("}", "\\}") | replace("&", "\\&") | replace("%", "\\%") | replace("$", "\\$") | replace("#", "\\#") | replace("_", "\\_") | replace("~", "\\textasciitilde{}") | replace("^", "\\textasciicircum{}")-][# if not loop.last #], [# endif #][# endfor #]}
[# endif #]

[# endif #]

[# endif #]

[# for affiliation in doc.affiliations #]

%% empty suppresses the trailing comma that would otherwise appear after a
%% standalone organization name (no department/address/city/etc.).
\affiliation[[-affiliation.index-]]{organization={[-affiliation.name | replace("\\", "\\textbackslash{}") | replace("{", "\\{") | replace("}", "\\}") | replace("&", "\\&") | replace("%", "\\%") | replace("$", "\\$") | replace("#", "\\#") | replace("_", "\\_") | replace("~", "\\textasciitilde{}") | replace("^", "\\textasciicircum{}")-]}[# if not has_address #],op={}[# endif #][# if affiliation.department #],
            department={[-affiliation.department | replace("\\", "\\textbackslash{}") | replace("{", "\\{") | replace("}", "\\}") | replace("&", "\\&") | replace("%", "\\%") | replace("$", "\\$") | replace("#", "\\#") | replace("_", "\\_") | replace("~", "\\textasciitilde{}") | replace("^", "\\textasciicircum{}")-]}[# endif #][# if affiliation.address #],
            addressline={[-affiliation.address | replace("\\", "\\textbackslash{}") | replace("{", "\\{") | replace("}", "\\}") | replace("&", "\\&") | replace("%", "\\%") | replace("$", "\\$") | replace("#", "\\#") | replace("_", "\\_") | replace("~", "\\textasciitilde{}") | replace("^", "\\textasciicircum{}")-]}[# endif #][# if affiliation.city #],
            city={[-affiliation.city | replace("\\", "\\textbackslash{}") | replace("{", "\\{") | replace("}", "\\}") | replace("&", "\\&") | replace("%", "\\%") | replace("$", "\\$") | replace("#", "\\#") | replace("_", "\\_") | replace("~", "\\textasciitilde{}") | replace("^", "\\textasciicircum{}")-]}[# endif #][# if affiliation.postal_code #],
            postcode={[-affiliation.postal_code | replace("\\", "\\textbackslash{}") | replace("{", "\\{") | replace("}", "\\}") | replace("&", "\\&") | replace("%", "\\%") | replace("$", "\\$") | replace("#", "\\#") | replace("_", "\\_") | replace("~", "\\textasciitilde{}") | replace("^", "\\textasciicircum{}")-]}[# endif #][# if affiliation.state #],
            state={[-affiliation.state | replace("\\", "\\textbackslash{}") | replace("{", "\\{") | replace("}", "\\}") | replace("&", "\\&") | replace("%", "\\%") | replace("$", "\\$") | replace("#", "\\#") | replace("_", "\\_") | replace("~", "\\textasciitilde{}") | replace("^", "\\textasciicircum{}")-]}[# endif #][# if affiliation.country #],
            country={[-affiliation.country | replace("\\", "\\textbackslash{}") | replace("{", "\\{") | replace("}", "\\}") | replace("&", "\\&") | replace("%", "\\%") | replace("$", "\\$") | replace("#", "\\#") | replace("_", "\\_") | replace("~", "\\textasciitilde{}") | replace("^", "\\textasciicircum{}")-]}[# endif #]}
[# endfor #]
